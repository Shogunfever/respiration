    <script>
        // Audio context for sounds
        let audioContext = null;
        let isRunning = false;
        let timerInterval = null;
        let currentPhase = 'idle';
        let phaseTimer = null;
        
        // Breathing phases
        const phases = ['inspire', 'hold_inspire', 'expire', 'hold_expire'];
        const phaseLabels = {
            'inspire': 'Inspiration',
            'hold_inspire': 'Retenue',
            'expire': 'Expiration',
            'hold_expire': 'Retenue'
        };
        
        // Initialize audio context on first interaction
        function initAudio() {
            if (!audioContext) {
                audioContext = new (window.AudioContext || window.webkitAudioContext)();
            }
        }
        
        // Play notification sound for phase change
        function playNotification(phase) {
            if (!document.getElementById('soundToggle').checked) return;
            
            initAudio();
            
            const oscillator = audioContext.createOscillator();
            const gainNode = audioContext.createGain();
            
            oscillator.connect(gainNode);
            gainNode.connect(audioContext.destination);
            
            // Different tones for different phases
            const frequencies = {
                'inspire': 440,      // A4
                'hold_inspire': 523, // C5
                'expire': 392,       // G4
                'hold_expire': 349   // F4
            };
            
            oscillator.frequency.value = frequencies[phase] || 440;
            oscillator.type = 'sine';
            
            gainNode.gain.setValueAtTime(0.3, audioContext.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.01, audioContext.currentTime + 0.5);
            
            oscillator.start(audioContext.currentTime);
            oscillator.stop(audioContext.currentTime + 0.5);
        }
        
        // Update value displays
        function updateValues() {
            document.getElementById('inspireValue').textContent = document.getElementById('inspireTime').value + 's';
            document.getElementById('holdInspireValue').textContent = document.getElementById('holdInspireTime').value + 's';
            document.getElementById('expireValue').textContent = document.getElementById('expireTime').value + 's';
            document.getElementById('holdExpireValue').textContent = document.getElementById('holdExpireTime').value + 's';
        }
        
        // Load presets
        function loadPreset(name) {
            document.querySelectorAll('.preset-btn').forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');
            
            switch(name) {
                case 'coherence':
                    setTimes(4, 0, 4, 0);
                    break;
                case 'relax':
                    setTimes(4, 0, 7, 8);
                    break;
                case 'box':
                    setTimes(4, 4, 4, 4);
                    break;
            }
        }
        
        function setTimes(inspire, holdInspire, expire, holdExpire) {
            document.getElementById('inspireTime').value = inspire;
            document.getElementById('holdInspireTime').value = holdInspire;
            document.getElementById('expireTime').value = expire;
            document.getElementById('holdExpireTime').value = holdExpire;
            updateValues();
        }
        
        // Breathing cycle
        async function breatheCycle() {
            if (!isRunning) return;
            
            // Inspiration
            await runPhase('inspire', parseInt(document.getElementById('inspireTime').value));
            if (!isRunning) return;
            
            // Hold inspiration
            await runPhase('hold_inspire', parseInt(document.getElementById('holdInspireTime').value));
            if (!isRunning) return;
            
            // Expiration
            await runPhase('expire', parseInt(document.getElementById('expireTime').value));
            if (!isRunning) return;
            
            // Hold expiration
            await runPhase('hold_expire', parseInt(document.getElementById('holdExpireTime').value));
            if (!isRunning) return;
            
            // Repeat
            breatheCycle();
        }
        
        async function runPhase(phase, duration) {
            if (duration === 0) return;
            
            currentPhase = phase;
            const circle = document.getElementById('circle');
            const phaseText = document.getElementById('phaseText');
            const timerDisplay = document.getElementById('timerDisplay');
            
            // Update visual
            circle.className = 'breathing-circle';
            void circle.offsetWidth; // Trigger reflow
            
            if (phase === 'inspire') {
                circle.classList.add('inspire');
            } else if (phase === 'expire') {
                circle.classList.add('expire');
            }
            
            phaseText.textContent = phaseLabels[phase];
            playNotification(phase);
            
            // Countdown timer
            let remaining = duration;
            timerDisplay.textContent = remaining;
            
            return new Promise(resolve => {
                timerInterval = setInterval(() => {
                    remaining--;
                    timerDisplay.textContent = remaining;
                    
                    if (remaining <= 0) {
                        clearInterval(timerInterval);
                        resolve();
                    }
                }, 1000);
            });
        }
        
        function startBreathing() {
            initAudio();
            isRunning = true;
            document.getElementById('startBtn').disabled = true;
            document.getElementById('stopBtn').disabled = false;
            breatheCycle();
        }
        
        function stopBreathing() {
            isRunning = false;
            clearInterval(timerInterval);
            clearTimeout(phaseTimer);
            document.getElementById('startBtn').disabled = false;
            document.getElementById('stopBtn').disabled = true;
            document.getElementById('phaseText').textContent = 'Arrêté';
            document.getElementById('timerDisplay').textContent = '0';
            document.getElementById('circle').className = 'breathing-circle';
        }
        
        // Initialize
        updateValues();
    </script>
